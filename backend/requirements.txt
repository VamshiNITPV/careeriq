# --- security ---
# PyJWT rather than python-jose: actively maintained, smaller surface, and the
# only thing we need is HS256 sign/verify.
pyjwt>=2.10
# bcrypt directly rather than passlib: passlib is effectively unmaintained and
# its bcrypt backend misdetects the version of bcrypt 4.x, emitting warnings and
# in some versions failing outright.
bcrypt>=4.2

# --- document parsing ---
# pdfplumber over PyPDF2: it preserves word positions and layout, which is what
# makes two-column resumes extractable rather than interleaved nonsense.
pdfplumber>=0.11
python-docx>=1.1

# --- email ---
# Async SMTP. stdlib smtplib is blocking and would stall the event loop for the
# duration of every send (ADR-004).
aiosmtplib>=3.0

# --- misc ---
structlog>=24.4
python-multipart>=0.0.20
